ALGORITHME / calculateur billet sncf


VARIABLES:
	age : ENTIER
	a_carte_reduction : BOOLEEN
	prix : REEL
CONSTANTE :
	PRIX_BASE = 100.0


DEBUT:
	AFFICHER("Bonjour ! Entrez l'âge du voyageur)
	CHOIX(age)
	AFFICHER("Le voyageur possède-t-il une carte de réduction (VRAI/FAUX)")
	CHOIX(a_carte_reduction)

	SI age <= 12 ALORS:
		prix <- 0.0
	SINON SI age <=25 ET age >= 12 ALORS:
		prix <- PRIX_BASE * 0.50
		SI a_carte_reduction == VRAI ALORS:
			prix <- prix * 0.90
	SINON:
		prix <- PRIX_BASE
		SI a_carte_reduction == VRAI ALORS:
			prix <- prix * 0.80
	FIN SI
FIN
